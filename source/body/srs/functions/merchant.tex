\subsection{商家用例}

\srsfigure{figures/srs/12-merchant-use-case.pdf}{商家用例图}{UML 用例图}{fig:merchant-use-case}

\begin{srsusecase}{UC-MER-01}{商家申请与店铺状态}{顾客、已审核商家}{P0}{uc:merchant-store}
\ucrow{基本流程}{1）用户提交商家资质申请；2）通过后提交店铺基本信息；3）店铺审核通过后，商家可在营业、休息和临时闭店之间切换；4）顾客端同步展示状态。}
\ucrow{异常流程}{未审核用户不得进入商家工作台；非店铺所有者、未审核店铺或非法状态值的修改请求被拒绝。}
\ucrow{验收标准}{商家和店铺审核层次独立；非营业店铺不能创建新订单；商家 A 不得修改商家 B 的店铺。}
\end{srsusecase}

\begin{srsusecase}{UC-MER-02}{分类、商品与库存管理}{已审核商家}{P0}{uc:product-stock}
\ucrow{基本流程}{1）查看本人店铺分类和商品；2）新增、修改和删除未被有效订单引用的分类；3）新增、修改、上下架商品；4）维护价格、库存、限购、图片和标签。}
\ucrow{异常流程}{价格小于等于零、库存小于零、限购非正整数、非本店资源或删除已被订单引用的商品时拒绝操作。}
\ucrow{说明}{商品被历史订单引用后使用逻辑下架而非物理删除；库存预占和恢复必须通过统一业务服务。}
\ucrow{验收标准}{下架、售罄和超限购商品不可下单；两个并发订单不得使库存为负。}
\end{srsusecase}

\begin{srsusecase}{UC-MER-03}{商家处理订单}{订单所属店铺的商家}{P0}{uc:merchant-order}
\ucrow{基本流程}{1）按状态查看本店订单；2）对待接单订单执行接单或拒单；3）接单后进入制作中；4）制作完成后确认出餐；5）系统创建待骑手接单的配送任务。}
\ucrow{异常流程}{非本店订单、已取消订单、重复接单/出餐或跳过制作状态时拒绝操作。}
\ucrow{验收标准}{接单、拒单、出餐严格遵循表~\ref{tab:order-state}；重复出餐不会创建第二条活动配送任务。}
\end{srsusecase}

\begin{srsusecase}{UC-MER-04}{评价回复与经营看板}{已审核商家}{P1}{uc:merchant-analytics}
\ucrow{基本流程}{1）查看本店可展示评价；2）对未回复评价提交一次回复；3）按时间查看已完成订单量、营业额、客单价、热销商品和评分趋势；4）可选获取 AI 生成的数据摘要。}
\ucrow{说明}{商家只能回复本店评价，不得修改顾客评分和原文；AI 摘要仅解释统计结果，不改变原始数据。}
\ucrow{验收标准}{取消订单不计入营业额；未授权用户无法查看店铺私有统计；回复后顾客原评价不变。}
\end{srsusecase}
