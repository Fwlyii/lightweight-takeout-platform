\subsubsection{8 管理店铺商品}

\begin{srsusecase}{8.1}{管理店铺信息}{商家}{高}{uc:merchant-store-product}
\ucrow{描述}{商家维护本人店铺资料和营业状态}
\ucrow{前置条件}{用户已通过商家角色审核}
\ucrow{基本流程}{1）提交店铺基本资料\newline 2）查看店铺审核状态\newline 3）店铺审核通过后维护店铺资料\newline 4）在营业、休息和临时闭店之间切换营业状态}
\ucrow{结束状况}{店铺资料或营业状态更新完成}
\ucrow{可选流程}{审核未通过时根据原因修改资料后重新提交}
\ucrow{异常流程}{未审核商家、非店铺所有者、店铺未通过审核或营业状态非法时拒绝操作}
\ucrow{备注}{商家角色审核和店铺审核相互独立，非营业店铺不能创建新订单}
\end{srsusecase}

\begin{srsusecase}{8.2}{管理商品分类}{商家}{高}{uc:category-manage}
\ucrow{描述}{商家维护本人店铺的商品分类}
\ucrow{前置条件}{商家已拥有可管理店铺}
\ucrow{基本流程}{1）查看本店商品分类\newline 2）新增商品分类\newline 3）修改分类名称或排序\newline 4）删除允许删除的分类}
\ucrow{结束状况}{商品分类信息更新完成}
\ucrow{可选流程}{可调整分类展示顺序}
\ucrow{异常流程}{非本店资源或分类已被有效订单相关商品引用且不允许删除时拒绝操作}
\ucrow{备注}{分类操作只允许作用于本人店铺}
\end{srsusecase}

\begin{srsusecase}{8.3}{管理商品库存}{商家}{高}{uc:product-stock}
\ucrow{描述}{商家维护商品信息、上下架状态、库存和限购}
\ucrow{前置条件}{商家已拥有可管理店铺}
\ucrow{基本流程}{1）查看本店商品\newline 2）新增或修改商品名称、价格、图片和标签\newline 3）设置商品上下架状态\newline 4）维护库存和限购数量}
\ucrow{结束状况}{商品及库存信息更新完成}
\ucrow{可选流程}{历史订单引用过的商品可采用逻辑下架方式停止销售}
\ucrow{异常流程}{价格小于等于零、库存小于零、限购非正整数或操作非本店商品时拒绝请求}
\ucrow{备注}{库存预占和恢复通过统一业务服务完成，并发订单不得使库存为负}
\end{srsusecase}

\subsubsection{9 处理商家订单}

\begin{srsusecase}{9.1}{查看本店订单}{商家}{高}{uc:merchant-order}
\ucrow{描述}{商家查看本人店铺收到的订单}
\ucrow{前置条件}{商家已通过审核并拥有当前店铺}
\ucrow{基本流程}{1）进入商家订单工作台\newline 2）按订单状态查看本店订单列表\newline 3）选择订单查看商品、金额和当前状态}
\ucrow{结束状况}{显示本人店铺订单列表或订单详情}
\ucrow{可选流程}{可按状态筛选待接单、制作中和已完成等订单}
\ucrow{异常流程}{访问非本人店铺订单时拒绝请求}
\ucrow{备注}{商家只能查看和处理本店订单}
\end{srsusecase}

\begin{srsusecase}{9.2}{处理接单请求}{商家}{高}{uc:merchant-accept}
\ucrow{描述}{商家对待接单订单选择接受或拒绝}
\ucrow{前置条件}{订单属于当前店铺且状态为待商家接单}
\ucrow{基本流程}{1）打开待接单订单\newline 2）选择接受或拒绝\newline 3）系统校验订单当前状态\newline 4）更新订单状态并通知顾客}
\ucrow{结束状况}{接受后订单进入制作中，拒绝后订单进入相应终止状态}
\ucrow{可选流程}{拒绝订单时可填写拒单原因}
\ucrow{异常流程}{非本店订单、已取消订单或重复接单拒单时拒绝操作}
\ucrow{备注}{接单与拒单结果必须符合订单状态转换规则}
\end{srsusecase}

\begin{srsusecase}{9.3}{确认订单出餐}{商家}{高}{uc:merchant-ready}
\ucrow{描述}{商家在餐品制作完成后确认出餐}
\ucrow{前置条件}{订单属于当前店铺且已经接受并进入制作阶段}
\ucrow{基本流程}{1）打开制作中订单\newline 2）确认餐品制作完成\newline 3）提交确认出餐\newline 4）系统更新订单状态并创建待骑手接取的配送任务}
\ucrow{结束状况}{订单进入等待配送阶段并生成配送任务}
\ucrow{可选流程}{无}
\ucrow{异常流程}{订单已取消、重复确认出餐或跳过合法状态时拒绝操作}
\ucrow{备注}{重复出餐不得创建第二条活动配送任务}
\end{srsusecase}

\subsubsection{10 管理评价与经营}

\begin{srsusecase}{10.1}{回复顾客评价}{商家}{中}{uc:merchant-operation}
\ucrow{描述}{商家查看并回复本人店铺的顾客评价}
\ucrow{前置条件}{商家已登录且评价属于本人店铺}
\ucrow{基本流程}{1）查看本店可展示评价\newline 2）选择尚未回复的评价\newline 3）填写回复内容\newline 4）提交回复}
\ucrow{结束状况}{评价显示商家回复内容}
\ucrow{可选流程}{无}
\ucrow{异常流程}{评价属于其他店铺、已经回复或当前评价不可展示时拒绝操作}
\ucrow{备注}{商家不能修改顾客评分和评价原文}
\end{srsusecase}

\begin{srsusecase}{10.2}{查看经营数据}{商家}{中}{uc:merchant-analytics}
\ucrow{描述}{商家查看本人店铺的经营统计}
\ucrow{前置条件}{商家已登录且拥有当前店铺}
\ucrow{基本流程}{1）选择统计时间范围\newline 2）系统统计已完成订单\newline 3）显示订单量、营业额、客单价、热销商品和评分趋势}
\ucrow{结束状况}{显示所选时间范围内的经营数据}
\ucrow{可选流程}{可获取 AI 经营摘要，对已有统计结果进行解释}
\ucrow{异常流程}{未授权访问、时间范围非法或 AI 服务失败时给出提示，AI 失败不影响原始统计展示}
\ucrow{备注}{取消订单不计入营业额，AI 摘要不修改订单、评价和原始统计数据}
\end{srsusecase}
