\subsubsection{13 审核准入申请}

\begin{srsusecase}{13.1}{审核商家申请}{管理员}{高}{uc:admin-audit}
\ucrow{描述}{管理员审核用户提交的商家资格申请}
\ucrow{前置条件}{存在待审核商家申请}
\ucrow{基本流程}{1）查看待审核商家申请\newline 2）查看申请人和资质资料\newline 3）选择通过或拒绝\newline 4）拒绝时填写原因\newline 5）系统保存审核结果并通知申请人}
\ucrow{结束状况}{商家申请进入已通过或已拒绝状态}
\ucrow{可选流程}{审核通过后授予用户商家角色}
\ucrow{异常流程}{申请已撤回、已经审核或拒绝原因为空时拒绝提交}
\ucrow{备注}{每次审核保存原状态、审核结果、管理员、原因和时间}
\end{srsusecase}

\begin{srsusecase}{13.2}{审核店铺申请}{管理员}{高}{uc:admin-store-audit}
\ucrow{描述}{管理员审核商家提交的店铺资料}
\ucrow{前置条件}{商家角色已通过审核且存在待审核店铺申请}
\ucrow{基本流程}{1）查看待审核店铺申请\newline 2）查看店铺基本资料\newline 3）选择通过或拒绝\newline 4）拒绝时填写原因\newline 5）系统保存审核结果并通知商家}
\ucrow{结束状况}{店铺申请进入已通过或已拒绝状态}
\ucrow{可选流程}{审核通过后允许商家维护营业状态和开展经营}
\ucrow{异常流程}{申请不存在、已经审核或店铺归属异常时拒绝操作}
\ucrow{备注}{店铺审核与商家角色审核相互独立}
\end{srsusecase}

\begin{srsusecase}{13.3}{审核骑手申请}{管理员}{高}{uc:admin-rider-audit}
\ucrow{描述}{管理员审核用户提交的骑手资格申请}
\ucrow{前置条件}{存在待审核骑手申请}
\ucrow{基本流程}{1）查看待审核骑手申请\newline 2）查看申请人资格资料\newline 3）选择通过或拒绝\newline 4）拒绝时填写原因\newline 5）系统保存审核结果并通知申请人}
\ucrow{结束状况}{骑手申请进入已通过或已拒绝状态}
\ucrow{可选流程}{审核通过后授予用户骑手角色}
\ucrow{异常流程}{申请已撤回、已经审核或拒绝原因为空时拒绝提交}
\ucrow{备注}{审核通过后骑手才能进入骑手工作台}
\end{srsusecase}

\subsubsection{14 维护平台秩序}

\begin{srsusecase}{14.1}{管理用户账号}{管理员}{高}{uc:admin-governance}
\ucrow{描述}{管理员查询并启用或禁用平台用户账号}
\ucrow{前置条件}{管理员已登录}
\ucrow{基本流程}{1）按条件查询用户账号\newline 2）查看账号当前状态\newline 3）选择启用或禁用\newline 4）系统保存状态变化}
\ucrow{结束状况}{用户账号更新为新的可用状态}
\ucrow{可选流程}{可填写治理原因}
\ucrow{异常流程}{账号不存在、状态不允许变更或普通用户调用管理接口时拒绝请求}
\ucrow{备注}{治理操作保留原账号及历史业务数据}
\end{srsusecase}

\begin{srsusecase}{14.2}{管理违规评价}{管理员}{高}{uc:admin-review-governance}
\ucrow{描述}{管理员隐藏或恢复平台中的违规评价}
\ucrow{前置条件}{管理员已登录且目标评价存在}
\ucrow{基本流程}{1）查看目标评价及当前展示状态\newline 2）选择隐藏或恢复展示\newline 3）填写治理原因\newline 4）系统保存评价展示状态变化}
\ucrow{结束状况}{评价按照新的治理状态展示}
\ucrow{可选流程}{无}
\ucrow{异常流程}{评价不存在、状态不允许变更或普通用户调用管理接口时拒绝请求}
\ucrow{备注}{治理操作不删除顾客原评价内容和历史记录}
\end{srsusecase}

\begin{srsusecase}{14.3}{处置配送异常}{管理员}{高}{uc:admin-delivery-exception}
\ucrow{描述}{管理员处理骑手上报的异常配送任务}
\ucrow{前置条件}{存在待处理配送异常}
\ucrow{基本流程}{1）查看异常配送详情和骑手说明\newline 2）判断当前任务状态\newline 3）选择恢复配送、改派配送或终止配送\newline 4）填写处理说明\newline 5）系统更新任务状态并通知相关用户}
\ucrow{结束状况}{异常配送进入新的可执行或终止状态}
\ucrow{可选流程}{改派配送时重新选择符合条件的骑手}
\ucrow{异常流程}{任务不存在、状态不允许处理或重复提交治理结果时拒绝请求}
\ucrow{备注}{终止配送时同步处理订单状态和相关资产}
\end{srsusecase}

\subsubsection{15 查看平台数据}

\begin{srsusecase}{15.1}{查看平台概览}{管理员}{中}{uc:admin-dashboard}
\ucrow{描述}{管理员查看平台整体运行数据}
\ucrow{前置条件}{管理员已登录}
\ucrow{基本流程}{1）进入平台数据看板\newline 2）选择日期范围\newline 3）系统统计用户、商家、店铺、骑手和订单数量\newline 4）显示平台概览}
\ucrow{结束状况}{显示所选日期范围内的平台概览数据}
\ucrow{可选流程}{可切换日期范围重新统计}
\ucrow{异常流程}{日期范围非法或无权限访问时拒绝请求}
\ucrow{备注}{概览数据与相同条件的业务查询口径保持一致}
\end{srsusecase}

\begin{srsusecase}{15.2}{查看交易统计}{管理员}{中}{uc:admin-trade-stats}
\ucrow{描述}{管理员查看平台订单和营业额等交易统计}
\ucrow{前置条件}{管理员已登录}
\ucrow{基本流程}{1）进入交易统计页面\newline 2）选择日期范围\newline 3）系统统计已完成有效订单\newline 4）显示营业额、订单趋势和相关交易指标}
\ucrow{结束状况}{显示所选日期范围内的交易统计}
\ucrow{可选流程}{可按不同时间范围查看趋势变化}
\ucrow{异常流程}{日期范围非法或统计条件不合法时提示重新选择}
\ucrow{备注}{已取消订单不计入营业额}
\end{srsusecase}

\begin{srsusecase}{15.3}{查看配送统计}{管理员}{中}{uc:admin-delivery-stats}
\ucrow{描述}{管理员查看平台配送完成和异常情况}
\ucrow{前置条件}{管理员已登录}
\ucrow{基本流程}{1）进入配送统计页面\newline 2）选择日期范围\newline 3）系统统计配送任务数量、完成数量和异常数量\newline 4）计算并显示配送异常率}
\ucrow{结束状况}{显示所选日期范围内的配送统计}
\ucrow{可选流程}{可切换日期范围查看配送趋势}
\ucrow{异常流程}{日期范围非法或统计条件不合法时提示重新选择}
\ucrow{备注}{配送异常率的分母为所选期间内进入配送阶段的任务数}
\end{srsusecase}
