\subsection{管理员用例}

\begin{srsusecase}{13}{审核准入申请}{管理员}{P0}{uc:admin-audit}
\ucrow{用例分解}{13.1 审核商家申请；13.2 审核店铺申请；13.3 审核骑手申请。}
\ucrow{基本流程}{1）管理员按申请类型和状态查看待审核列表；2）查看申请人或店铺资料详情；3）选择通过或拒绝；4）拒绝时填写 1--200 字原因；5）系统更新审核状态、授予相应权限并通知申请人。}
\ucrow{异常流程}{普通用户访问、重复审核、审核已撤回申请或拒绝原因为空时拒绝操作。}
\ucrow{说明}{商家角色审核与店铺审核相互独立；骑手角色只有审核通过后才可进入骑手工作台。}
\ucrow{验收标准}{每次审核保存对象、原状态、结果、管理员、原因和时间；重复提交不得重复授权。}
\end{srsusecase}

\begin{srsusecase}{14}{维护平台秩序}{管理员}{P0}{uc:admin-governance}
\ucrow{用例分解}{14.1 管理用户账号；14.2 管理违规评价；14.3 处置配送异常。}
\ucrow{基本流程}{1）管理员按条件查询用户并启用或禁用账号；2）对违规评价执行隐藏或恢复展示；3）查看异常配送详情；4）根据业务情况选择恢复配送、改派配送或终止配送，并填写处理说明。}
\ucrow{异常流程}{普通用户调用管理接口、重复执行不允许的治理操作、处理不存在或状态不匹配的对象时拒绝请求。}
\ucrow{说明}{治理操作只改变展示或可用状态，不删除原评价、订单和配送审计数据；终止配送时必须按照订单状态规则同步处理订单与相关资产。}
\ucrow{验收标准}{普通用户调用管理接口返回 403；每次治理保存操作人、原状态、新状态、原因和时间；原始业务记录可追踪。}
\end{srsusecase}

\begin{srsusecase}{15}{查看平台数据}{管理员}{P1}{uc:admin-dashboard}
\ucrow{用例分解}{15.1 查看平台概览；15.2 查看交易统计；15.3 查看配送统计。}
\ucrow{基本流程}{1）管理员选择日期范围；2）查看用户、商家、店铺、骑手和订单等平台概览数据；3）查看已完成订单营业额、订单趋势等交易统计；4）查看配送完成数量、异常数量和异常率等配送统计；5）系统以数字卡片、趋势图和排名列表等方式展示。}
\ucrow{说明}{营业额只统计已完成有效订单；异常率的分母为所选期间内进入配送阶段的任务数。}
\ucrow{异常流程}{日期范围非法、无权限访问或统计条件不合法时拒绝请求并返回明确提示。}
\ucrow{验收标准}{平台概览、交易统计和配送统计与相同条件的业务查询口径一致；取消订单不计营业额。}
\end{srsusecase}
