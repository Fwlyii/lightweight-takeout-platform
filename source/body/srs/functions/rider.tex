\subsection{骑手用例}

\srsfigure{figures/srs/13-rider-use-case.pdf}{骑手用例图}{UML 用例图}{fig:rider-use-case}

\begin{srsusecase}{UC-RID-01}{骑手申请与工作状态}{顾客、已审核骑手}{P0}{uc:rider-profile}
\ucrow{基本流程}{1）用户提交骑手申请和必要资料；2）查看待审核、已通过或已拒绝结果；3）审核通过后进入骑手工作台；4）切换上线/下线状态。}
\ucrow{说明}{只有审核通过且上线的骑手可接新任务；存在活动任务时不得直接下线。}
\ucrow{验收标准}{未审核角色无法访问骑手接口；下线骑手不出现在新任务候选范围；活动任务未结束时下线失败。}
\end{srsusecase}

\begin{srsusecase}{UC-RID-02}{查看可接任务与并发接单}{已审核且上线骑手}{P0}{uc:rider-accept}
\ucrow{基本流程}{1）分页查看待接单任务的店铺、目的地大致区域、距离和任务状态；2）提交接单请求；3）系统校验身份、上线状态、任务状态和当前任务上限；4）原子绑定骑手并返回详情。}
\ucrow{异常流程}{未审核、已下线、已达任务上限、任务被他人接取或重复提交时返回可识别的冲突结果。}
\ucrow{验收标准}{两名合格骑手并发接取同一任务时恰好一人成功；失败请求不覆盖已建立的绑定。}
\end{srsusecase}

\begin{srsusecase}{UC-RID-03}{到店、取餐与送达}{当前任务绑定骑手}{P0}{uc:rider-deliver}
\ucrow{基本流程}{1）查看取餐地址并记录到店；2）商家已出餐后确认取餐；3）查看配送所需的收货地址和联系方式；4）开始配送；5）完成交付后确认送达。}
\ucrow{异常流程}{非绑定骑手、商家未出餐、跳过取餐直接送达或重复送达时拒绝操作。}
\ucrow{说明}{骑手只在活动履约期间读取必要配送信息；送达、取消或改派后失去对该任务敏感信息的访问权。}
\ucrow{验收标准}{状态顺序与表~\ref{tab:delivery-state} 一致；越权或越级请求无数据副作用；关键时间全部留存。}
\end{srsusecase}

\begin{srsusecase}{UC-RID-04}{配送异常、历史与导航}{当前任务骑手}{P0}{uc:rider-exception}
\ucrow{基本流程}{1）对无法联系顾客、地址错误、餐品洒损或其他异常选择类型并填写说明；2）任务进入异常待处理；3）查看管理员的恢复、改派或终止结果；4）查看本人历史任务；5）对活动任务打开外部地图导航。}
\ucrow{说明}{骑手不得自行将异常任务设为已取消；地图服务不可用时仍展示文本地址；历史任务不显示已失效的完整敏感信息。}
\ucrow{验收标准}{异常上报包含类型、说明、时间和骑手；处理结果可追踪；地图失败不阻断后续状态操作。}
\end{srsusecase}
