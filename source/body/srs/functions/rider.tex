\subsubsection{11 管理配送工作}

\begin{srsusecase}{11.1}{切换工作状态}{骑手}{高}{uc:rider-work}
\ucrow{描述}{骑手切换上线或下线工作状态}
\ucrow{前置条件}{用户已通过骑手资格审核并进入骑手工作台}
\ucrow{基本流程}{1）查看当前工作状态\newline 2）选择上线或下线\newline 3）系统校验当前活动任务限制\newline 4）更新骑手工作状态}
\ucrow{结束状况}{骑手处于新的上线或下线状态}
\ucrow{可选流程}{无}
\ucrow{异常流程}{存在不允许下线的活动任务时拒绝下线}
\ucrow{备注}{只有审核通过且上线的骑手进入新任务候选范围}
\end{srsusecase}

\begin{srsusecase}{11.2}{查看配送任务}{骑手}{高}{uc:rider-task-list}
\ucrow{描述}{骑手查看当前可接取的配送任务}
\ucrow{前置条件}{骑手已审核通过且处于上线状态}
\ucrow{基本流程}{1）进入可接任务页面\newline 2）系统读取待接单配送任务\newline 3）显示店铺、目的地区域、距离和任务状态\newline 4）骑手查看任务详情}
\ucrow{结束状况}{显示当前可接取的配送任务列表}
\ucrow{可选流程}{可分页查看更多任务}
\ucrow{异常流程}{骑手已下线或资格失效时不返回可接任务}
\ucrow{备注}{任务列表只显示当前仍可接取的配送任务}
\end{srsusecase}

\begin{srsusecase}{11.3}{接取配送任务}{骑手}{高}{uc:rider-accept}
\ucrow{描述}{骑手接取一项可用配送任务}
\ucrow{前置条件}{骑手已审核通过、处于上线状态且任务仍待接取}
\ucrow{基本流程}{1）选择配送任务\newline 2）提交接单请求\newline 3）系统校验骑手身份、工作状态、任务状态和当前任务上限\newline 4）系统原子绑定骑手并返回任务详情}
\ucrow{结束状况}{配送任务绑定当前骑手}
\ucrow{可选流程}{无}
\ucrow{异常流程}{已下线、达到任务上限、任务已被他人接取或重复提交时返回冲突结果}
\ucrow{备注}{两名合格骑手并发接取同一任务时只允许一人成功}
\end{srsusecase}

\subsubsection{12 执行配送任务}

\begin{srsusecase}{12.1}{确认取餐}{骑手}{高}{uc:rider-delivery}
\ucrow{描述}{骑手到店后确认取得待配送餐品}
\ucrow{前置条件}{配送任务已绑定当前骑手且商家已经出餐}
\ucrow{基本流程}{1）查看取餐店铺和地址\newline 2）需要时记录到店状态\newline 3）核对订单与餐品\newline 4）确认取餐并开始配送}
\ucrow{结束状况}{配送任务进入配送中状态}
\ucrow{可选流程}{可使用地图导航前往取餐店铺}
\ucrow{异常流程}{非绑定骑手、商家未出餐或重复确认取餐时拒绝操作}
\ucrow{备注}{地图服务不可用时仍显示文本地址}
\end{srsusecase}

\begin{srsusecase}{12.2}{确认送达}{骑手}{高}{uc:rider-delivered}
\ucrow{描述}{骑手完成配送后确认订单已送达}
\ucrow{前置条件}{当前任务已确认取餐并处于配送中状态}
\ucrow{基本流程}{1）查看收货地址和配送所需联系方式\newline 2）完成餐品交付\newline 3）提交确认送达\newline 4）系统记录送达时间并更新订单与配送状态}
\ucrow{结束状况}{配送任务进入已送达状态}
\ucrow{可选流程}{配送过程中可使用地图导航，发生无法联系顾客、地址错误或餐品洒损等情况时可上报配送异常}
\ucrow{异常流程}{非绑定骑手、跳过取餐直接送达、重复送达或异常任务越级操作时拒绝请求}
\ucrow{备注}{送达、取消或改派后骑手不再访问该任务的完整敏感信息}
\end{srsusecase}
