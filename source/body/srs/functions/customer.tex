\subsubsection{4 选购店铺商品}

\begin{srsusecase}{4.1}{浏览搜索商品}{顾客}{高}{uc:shopping}
\ucrow{描述}{顾客浏览店铺并搜索符合条件的商品}
\ucrow{前置条件}{无}
\ucrow{基本流程}{1）查看已审核店铺及营业状态\newline 2）进入店铺查看商品分类和商品列表\newline 3）按关键词、分类、价格或销量进行搜索筛选\newline 4）查看搜索结果}
\ucrow{结束状况}{显示符合条件的店铺商品列表}
\ucrow{可选流程}{公开店铺和商品信息可在未登录状态浏览}
\ucrow{异常流程}{没有符合条件的商品时显示空结果}
\ucrow{备注}{休息店铺可以展示，但不可新增购买}
\end{srsusecase}

\begin{srsusecase}{4.2}{查看商品详情}{顾客}{高}{uc:product-detail}
\ucrow{描述}{顾客查看商品的详细信息}
\ucrow{前置条件}{商品存在且允许展示}
\ucrow{基本流程}{1）从商品列表选择商品\newline 2）系统读取商品名称、价格、库存、销量、图片和标签\newline 3）显示商品详情}
\ucrow{结束状况}{显示选中商品的详细信息}
\ucrow{可选流程}{可从详情页直接加入购物车}
\ucrow{异常流程}{商品不存在或已失效时提示无法查看}
\ucrow{备注}{最终可购买状态以后端实时校验结果为准}
\end{srsusecase}

\begin{srsusecase}{4.3}{管理购物车}{顾客}{高}{uc:cart}
\ucrow{描述}{顾客维护当前店铺的待结算商品}
\ucrow{前置条件}{顾客已登录且商品当前可售}
\ucrow{基本流程}{1）将商品加入购物车\newline 2）修改商品数量或删除商品\newline 3）选择待结算商品\newline 4）查看估算小计\newline 5）需要时清空购物车}
\ucrow{结束状况}{购物车保存最新商品和数量}
\ucrow{可选流程}{切换店铺加购时先确认是否清空原店铺购物车}
\ucrow{异常流程}{商品下架、售罄、超过库存或限购数量时拒绝加购或修改}
\ucrow{备注}{正式下单时重新读取价格、库存和营业状态}
\end{srsusecase}

\subsubsection{5 管理订单}

\begin{srsusecase}{5.1}{创建订单}{顾客}{高}{uc:customer-order}
\ucrow{描述}{顾客根据购物车商品创建外卖订单}
\ucrow{前置条件}{购物车至少有一个选中商品且存在有效收货地址}
\ucrow{基本流程}{1）选择收货地址、会员权益、优惠券和积分\newline 2）系统试算原价、优惠和应付金额\newline 3）顾客确认提交订单\newline 4）系统重新校验店铺、商品、价格、库存、地址和优惠资产\newline 5）生成订单及商品快照}
\ucrow{结束状况}{生成一笔待支付订单}
\ucrow{可选流程}{可选择不使用优惠券、积分或会员权益}
\ucrow{异常流程}{店铺休息、商品失效、价格变化、库存不足、地址无效或优惠失效时不生成订单}
\ucrow{备注}{相同幂等键重复提交不得创建两个订单}
\end{srsusecase}

\begin{srsusecase}{5.2}{支付订单}{顾客}{高}{uc:pay-order}
\ucrow{描述}{顾客支付本人待支付订单}
\ucrow{前置条件}{订单属于当前顾客且状态为待支付}
\ucrow{基本流程}{1）进入订单支付页面\newline 2）选择模拟支付或钱包支付\newline 3）系统校验订单金额、支付状态和余额\newline 4）记录支付结果并更新订单状态}
\ucrow{结束状况}{订单进入待商家接单状态}
\ucrow{可选流程}{钱包支付时扣减钱包余额并记录流水}
\ucrow{异常流程}{余额不足、重复支付、订单已取消或订单状态不允许支付时拒绝请求}
\ucrow{备注}{重复支付请求不得产生重复扣款}
\end{srsusecase}

\begin{srsusecase}{5.3}{取消订单}{顾客}{高}{uc:cancel-order}
\ucrow{描述}{顾客在允许取消的阶段取消本人订单}
\ucrow{前置条件}{订单属于当前顾客且当前状态允许取消}
\ucrow{基本流程}{1）进入订单详情\newline 2）提交取消订单请求\newline 3）系统校验订单状态\newline 4）更新订单状态\newline 5）恢复库存并回退相关优惠资产}
\ucrow{结束状况}{订单进入已取消状态}
\ucrow{可选流程}{已使用钱包、积分或优惠券时生成对应反向流水}
\ucrow{异常流程}{订单已进入不允许顾客自助取消的阶段或重复取消时拒绝请求}
\ucrow{备注}{取消操作需要保持订单、库存和资产状态一致}
\end{srsusecase}

\begin{srsusecase}{5.4}{跟踪订单}{顾客}{高}{uc:track-order}
\ucrow{描述}{顾客查看本人订单状态和配送进度}
\ucrow{前置条件}{订单属于当前顾客}
\ucrow{基本流程}{1）进入本人订单列表\newline 2）选择订单查看详情\newline 3）查看订单、商家和配送关键时间线\newline 4）订单送达后确认收货}
\ucrow{结束状况}{显示订单当前状态和履约进度}
\ucrow{可选流程}{送达后确认收货并进入评价入口}
\ucrow{异常流程}{非订单所有者访问或订单状态异常时拒绝请求}
\ucrow{备注}{重复确认收货不得重复发放积分}
\end{srsusecase}

\subsubsection{6 管理评价与消费}

\begin{srsusecase}{6.1}{提交订单评价}{顾客}{中}{uc:review-consumption}
\ucrow{描述}{顾客对已完成的本人订单提交评价}
\ucrow{前置条件}{订单属于当前顾客、状态为已完成且尚未评价}
\ucrow{基本流程}{1）进入订单评价入口\newline 2）填写 1--5 分评分和评价文字\newline 3）需要时上传评价图片\newline 4）提交评价}
\ucrow{结束状况}{订单生成一条有效顾客评价}
\ucrow{可选流程}{最多上传 3 张符合要求的图片}
\ucrow{异常流程}{订单未完成、非本人订单、重复评价、评分越界或图片不符合要求时拒绝提交}
\ucrow{备注}{每个订单最多一条有效顾客评价}
\end{srsusecase}

\begin{srsusecase}{6.2}{查看消费分析}{顾客}{中}{uc:consumption-analysis}
\ucrow{描述}{顾客查看本人历史消费统计}
\ucrow{前置条件}{顾客已登录}
\ucrow{基本流程}{1）进入个人消费分析页面\newline 2）选择月份或年份\newline 3）系统统计已完成订单\newline 4）显示消费金额、订单次数、店铺和品类等结果}
\ucrow{结束状况}{显示所选时间范围内的个人消费分析}
\ucrow{可选流程}{可查看商家对本人历史评价的回复}
\ucrow{异常流程}{查询条件不合法时提示重新选择}
\ucrow{备注}{已取消订单不进入消费统计，用户不得查询他人消费数据}
\end{srsusecase}

\subsubsection{7 使用智能与个性化服务}

\begin{srsusecase}{7.1}{使用 AI 点餐}{顾客}{中}{uc:intelligent-personalized}
\ucrow{描述}{顾客通过自然语言获取平台问答、订单查询和点餐建议}
\ucrow{前置条件}{平台规则问答可直接使用，查询本人订单和确认加购时需要登录}
\ucrow{基本流程}{1）顾客输入平台规则、订单状态或点餐需求\newline 2）系统识别用户意图\newline 3）规则问答读取审核知识，订单查询校验用户归属，点餐请求查询真实可售商品\newline 4）返回文本答复或推荐商品卡片\newline 5）顾客确认推荐后加入购物车}
\ucrow{结束状况}{返回可理解答复或可继续操作的商品推荐结果}
\ucrow{可选流程}{可使用图片识菜或语音输入，识别结果由顾客确认后再进入受控搜索}
\ucrow{异常流程}{AI 超时、输出无法解析、无候选商品或识别失败时提示用户并降级到普通搜索}
\ucrow{备注}{AI 不直接支付、取消订单、修改库存或调整用户资产}
\end{srsusecase}

\begin{srsusecase}{7.2}{管理个人权益}{顾客}{中}{uc:personal-assets}
\ucrow{描述}{顾客查看和使用钱包、积分、优惠券与会员权益}
\ucrow{前置条件}{顾客已登录}
\ucrow{基本流程}{1）查看钱包、积分、优惠券、会员状态和变更流水\newline 2）需要时进行教学演示用模拟充值或开通会员\newline 3）结算时选择可用优惠券和积分\newline 4）支付成功后完成权益核销}
\ucrow{结束状况}{个人权益余额和流水更新完成}
\ucrow{可选流程}{订单取消时生成反向流水恢复相应权益}
\ucrow{异常流程}{余额不足、积分不足、优惠券过期或已使用、重复核销或并发冲突时拒绝操作}
\ucrow{备注}{每单最多使用一张优惠券，100 积分抵 1 元且不超过券后金额的 20\%}
\end{srsusecase}

\begin{srsusecase}{7.3}{管理个性化偏好}{顾客}{中}{uc:preference}
\ucrow{描述}{顾客维护口味偏好和个性化展示设置}
\ucrow{前置条件}{顾客已登录}
\ucrow{基本流程}{1）设置辣度、口味和忌口标签\newline 2）保存个人偏好\newline 3）搜索和 AI 推荐读取当前偏好\newline 4）需要时切换界面主题或清除偏好}
\ucrow{结束状况}{新的个性化偏好保存并在后续会话中恢复}
\ucrow{可选流程}{可启用每日主题，也可随时关闭}
\ucrow{异常流程}{偏好值不合法时提示重新选择}
\ucrow{备注}{个性化偏好只影响展示、搜索和推荐，不改变历史订单}
\end{srsusecase}
