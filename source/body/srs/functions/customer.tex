\subsection{顾客用例}

\srsfigure{figures/srs/11-customer-use-case.png}{顾客用例图}{UML 用例图}{fig:customer-use-case}

\begin{srsusecase}{UC-CUS-01}{浏览与搜索店铺商品}{游客、顾客}{P0}{uc:browse-search}
\ucrow{基本流程}{1）分页查看已审核店铺及营业状态；2）进入店铺查看分类、商品、价格、库存和销量；3）按关键词、分类、价格或销量筛选排序；4）进入商品详情。}
\ucrow{异常流程}{查询无结果时返回空状态；休息店铺可展示但不可新增购物车项；下架、售罄或超过限购的商品不可购买。}
\ucrow{验收标准}{分页结果不重复不遗漏；关键词和筛选组合结果正确；不可售商品的后端加购请求被拒绝。}
\end{srsusecase}

\begin{srsusecase}{UC-CUS-02}{管理购物车}{已登录顾客}{P0}{uc:cart}
\ucrow{基本流程}{1）将当前店铺可售商品加入购物车；2）修改数量或删除单项；3）选中待结算项；4）查看估算小计；5）清空购物车。}
\ucrow{说明}{同一商品重复加入时合并数量；切换店铺加购时要求顾客确认清空原店铺购物车；正式下单时重新读取价格、库存和营业状态。}
\ucrow{异常流程}{数量小于 1、超过库存/限购、商品失效或越权修改他人购物车时拒绝操作。}
\ucrow{验收标准}{增、改、删、清空及合并数量结果正确；购物车估算金额不被当作最终订单金额接收。}
\end{srsusecase}

\begin{srsusecase}{UC-CUS-03}{订单试算与创建}{已登录顾客}{P0}{uc:create-order}
\ucrow{前置条件}{购物车至少有一个选中商品，存在有效收货地址。}
\ucrow{基本流程}{1）选择地址、会员权益、优惠券和积分；2）后端返回原价、各类优惠和应付金额；3）顾客确认提交；4）后端再次校验所有条件，在事务中预占库存/优惠券、生成订单与快照；5）返回待支付订单。}
\ucrow{异常流程}{店铺休息、商品下架、价格变动、库存不足、地址无效、优惠失效或重复提交时不得生成半完成订单。}
\ucrow{验收标准}{篡改前端应付金额不影响后端计算；创建失败后订单、库存和资产状态一致；相同幂等键重试不会创建两个订单。}
\end{srsusecase}

\begin{srsusecase}{UC-CUS-04}{支付与取消订单}{订单所有者}{P0}{uc:pay-cancel}
\ucrow{基本流程}{1）对待支付订单选择模拟支付或钱包支付；2）系统校验金额、支付状态和余额；3）记录支付结果并将订单置为待商家接单；4）待支付或待商家接单时可提交取消；5）系统执行状态变更、库存恢复和资产回退。}
\ucrow{异常流程}{余额不足、重复支付、已取消订单支付、制作开始后顾客自助取消或越权操作时拒绝请求。}
\ucrow{验收标准}{连续两次支付/取消不产生重复副作用；取消后库存、优惠券、积分和钱包流水与表~\ref{tab:order-state} 一致。}
\end{srsusecase}

\begin{srsusecase}{UC-CUS-05}{查看履约进度与确认收货}{订单所有者}{P0}{uc:track-confirm}
\ucrow{基本流程}{1）按状态和时间查看本人订单；2）查看订单、商家和配送关键时间线；3）送达后提交确认收货；4）系统将订单置为已完成，发放积分并开放评价入口。}
\ucrow{异常流程}{非订单所有者、订单未送达、已取消或重复确认时拒绝操作。}
\ucrow{验收标准}{顾客只能查看本人订单；订单与配送时间线顺序正确；重复确认不重复发放积分。}
\end{srsusecase}

\begin{srsusecase}{UC-CUS-06}{评价与个人消费分析}{已登录顾客}{P1}{uc:review-consumption}
\ucrow{基本流程}{1）对已完成本人订单提交 1--5 分、文字和最多 3 张图片；2）查看商家回复；3）按月份或年份查看已完成订单的金额、次数、店铺和品类统计。}
\ucrow{异常流程}{未完成、非本人、重复评价、评分越界、文字超过 500 字或图片数量/类型不合法时拒绝提交。}
\ucrow{验收标准}{每个订单最多一条有效评价；已取消订单不进入消费统计；用户无法查询他人消费数据。}
\end{srsusecase}
