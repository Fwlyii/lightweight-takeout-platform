\subsection{公共功能用例}

\begin{srsusecase}{UC-COM-01}{注册与登录}{游客、已注册用户}{P0}{uc:register-login}
\ucrow{前置条件}{游客提交注册或登录表单；注册时手机号尚未存在，登录时账号已注册且未被禁用。}
\ucrow{基本流程}{1）用户输入手机号和密码；2）系统校验字段格式；3）注册时检查唯一性并使用安全哈希保存密码；4）登录时校验凭证与账号状态；5）返回认证信息及角色列表。}
\ucrow{异常流程}{手机号非 11 位数字、密码长度不在 8--32 位、重复手机号、错误密码或禁用账号时拒绝请求，不泄露密码和内部异常。}
\ucrow{结束状况}{注册成功后建立顾客身份；登录成功后建立可验证的用户会话。}
\ucrow{验收标准}{至少覆盖成功注册、重复注册、非法手机号、过短密码、正确登录、错误密码和禁用账号七个场景。}
\end{srsusecase}

\begin{srsusecase}{UC-COM-02}{个人资料与配送地址}{已登录用户}{P0}{uc:profile-address}
\ucrow{前置条件}{用户进入个人中心或地址簿。}
\ucrow{基本流程}{1）查看并修改昵称、头像和联系信息；2）新增、编辑、删除配送地址；3）设置唯一默认地址；4）系统对每次请求校验当前用户与资源归属。}
\ucrow{异常流程}{收货人、手机号或详细地址为空，访问他人地址，或删除不存在地址时返回明确错误。}
\ucrow{说明}{历史订单保存收货信息快照，后续修改地址不得改变已生成订单。}
\ucrow{验收标准}{默认地址始终最多一条；用户 A 无法读写用户 B 的地址；修改地址后历史订单快照不变。}
\end{srsusecase}

\begin{srsusecase}{UC-COM-03}{角色切换与状态通知}{具有多角色的用户}{P1}{uc:role-notification}
\ucrow{基本流程}{1）用户查看本人已授予的角色；2）选择角色后进入对应工作台；3）订单、审核或配送发生关键变化时，系统向相关用户生成站内通知；4）用户可查看并标记已读。}
\ucrow{异常流程}{用户不得选择未授予或未通过审核的角色；实时连接失败时仍可通过通知列表查询最新消息。}
\ucrow{验收标准}{角色路由与后端权限一致；新订单、审核结果和配送送达至少生成一条可追踪通知。}
\end{srsusecase}
