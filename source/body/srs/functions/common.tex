\subsubsection{1 管理账号}

\begin{srsusecase}{1.1}{注册账号}{用户}{高}{uc:account}
\ucrow{描述}{用户使用手机号和密码注册平台账号}
\ucrow{前置条件}{手机号尚未注册}
\ucrow{基本流程}{1）填写手机号和密码\newline 2）系统校验手机号格式、密码长度和手机号唯一性\newline 3）系统保存账号信息并建立默认顾客身份\newline 4）返回注册成功结果}
\ucrow{结束状况}{平台生成可正常登录的用户账号}
\ucrow{可选流程}{无}
\ucrow{异常流程}{手机号格式错误、密码不符合要求或手机号已存在时提示用户修改}
\ucrow{备注}{密码采用安全哈希方式保存}
\end{srsusecase}

\begin{srsusecase}{1.2}{登录账号}{用户}{高}{uc:login}
\ucrow{描述}{用户使用已注册账号登录平台}
\ucrow{前置条件}{账号已注册且未被禁用}
\ucrow{基本流程}{1）输入手机号和密码\newline 2）系统校验登录凭证和账号状态\newline 3）系统读取用户已拥有的业务角色\newline 4）建立登录会话并进入平台}
\ucrow{结束状况}{用户获得有效登录会话}
\ucrow{可选流程}{用户拥有多个业务角色时可在登录后选择进入对应工作台}
\ucrow{异常流程}{账号不存在、密码错误或账号被禁用时提示登录失败}
\ucrow{备注}{登录失败时不返回密码或内部异常信息}
\end{srsusecase}

\begin{srsusecase}{1.3}{管理个人资料与地址}{用户}{高}{uc:profile-address}
\ucrow{描述}{用户维护个人资料和本人配送地址}
\ucrow{前置条件}{用户已登录}
\ucrow{基本流程}{1）进入个人中心查看昵称、头像和联系方式\newline 2）修改并保存个人资料\newline 3）进入地址管理查看本人地址\newline 4）新增、修改、删除地址或设置默认地址}
\ucrow{结束状况}{个人资料或地址信息更新成功}
\ucrow{可选流程}{用户可将任意有效地址设置为唯一默认地址}
\ucrow{异常流程}{地址字段不完整、访问他人地址或操作不存在地址时拒绝请求}
\ucrow{备注}{历史订单保存收货信息快照，后续修改地址不影响已生成订单}
\end{srsusecase}

\subsubsection{2 管理业务角色}

\begin{srsusecase}{2.1}{申请成为商家}{用户}{高}{uc:business-role}
\ucrow{描述}{用户提交商家资格申请}
\ucrow{前置条件}{用户已登录且尚未拥有已审核商家角色}
\ucrow{基本流程}{1）进入商家申请入口\newline 2）填写商家资格资料并提交\newline 3）系统保存申请并显示审核状态\newline 4）管理员审核后系统通知申请结果}
\ucrow{结束状况}{申请进入待审核、已通过或已拒绝状态}
\ucrow{可选流程}{申请通过后继续提交店铺资料并进入独立店铺审核}
\ucrow{异常流程}{资料不完整或重复提交有效申请时拒绝提交}
\ucrow{备注}{商家角色审核与店铺审核相互独立}
\end{srsusecase}

\begin{srsusecase}{2.2}{申请成为骑手}{用户}{高}{uc:rider-apply}
\ucrow{描述}{用户提交骑手资格申请}
\ucrow{前置条件}{用户已登录且尚未拥有已审核骑手角色}
\ucrow{基本流程}{1）进入骑手申请入口\newline 2）填写骑手资格资料并提交\newline 3）系统保存申请并显示审核状态\newline 4）管理员审核后系统通知申请结果}
\ucrow{结束状况}{申请进入待审核、已通过或已拒绝状态}
\ucrow{可选流程}{审核通过后可进入骑手工作台}
\ucrow{异常流程}{资料不完整或重复提交有效申请时拒绝提交}
\ucrow{备注}{只有审核通过的骑手才能切换工作状态和接取任务}
\end{srsusecase}

\begin{srsusecase}{2.3}{切换业务角色}{用户}{高}{uc:role-switch}
\ucrow{描述}{用户在本人已获得授权的业务角色之间切换}
\ucrow{前置条件}{用户已登录且至少拥有一个可用业务角色}
\ucrow{基本流程}{1）查看本人已获得授权的角色\newline 2）选择目标业务角色\newline 3）系统校验角色权限\newline 4）进入对应工作台}
\ucrow{结束状况}{当前会话切换至目标业务角色}
\ucrow{可选流程}{用户可再次切换回其他已授权角色}
\ucrow{异常流程}{选择未授权或尚未审核通过的角色时拒绝切换}
\ucrow{备注}{角色切换结果与后端权限保持一致}
\end{srsusecase}

\subsubsection{3 查看消息通知}

\begin{srsusecase}{3}{查看消息通知}{用户}{中}{uc:notification}
\ucrow{描述}{用户查看与本人业务相关的站内通知}
\ucrow{前置条件}{用户已登录}
\ucrow{基本流程}{1）进入通知列表\newline 2）按时间查看本人通知\newline 3）打开通知查看关联业务信息\newline 4）将通知标记为已读}
\ucrow{结束状况}{显示本人通知列表及阅读状态}
\ucrow{可选流程}{可从通知进入对应订单、审核或配送详情}
\ucrow{异常流程}{实时连接失败时通过通知列表重新获取最新消息}
\ucrow{备注}{用户不能读取或修改他人的通知}
\end{srsusecase}
